**Tech Stack:** Python 3.12 for exports; Astro 7.4 (devDependencies also @astrojs/check and @types/node), TypeScript strict, Node 24, npm; Playwright for checks. No UI framework, no CSS framework.

```python
# ---- head to head (2026 series between Worlds teams, plus 2025 internationals) ----
WORLDS = [r["code"] for r in rat["teams"]]
tm = J("data/team_map.json"); name2code = {n: v["code"] for n, v in tm.items()}
alias25 = {"Gen.G eSports": "GEN", "Gen.G Esports": "GEN", "Hanwha Life eSports": "HLE", "Anyone's Legend": "AL", "Team Liquid Honda": "TLAW", "FURIA Esports": "FURIA"}
def _code(n): return name2code.get(n) or alias25.get(n)
def _year(d): return int(d.split("/")[2])
h2h = []
for r in J("data/series_2026.json"):
    if r["a"] in WORLDS and r["b"] in WORLDS:
        a, b, sa, sb = (r["a"], r["b"], r["sa"], r["sb"]) if r["a"] < r["b"] else (r["b"], r["a"], r["sb"], r["sa"])
        h2h.append(dict(a=a, b=b, date=r["date"], tournament=r["tournament"], stage=r["stage"], sa=sa, sb=sb, bo=r["bo"], patch=r.get("patch"), year=_year(r["date"])))
for r in J("data/golgg_intl_2025.json"):
    ca, cb = _code(r["team_a"]), _code(r["team_b"])
    if ca in WORLDS and cb in WORLDS:
        a, b, sa, sb = (ca, cb, r["score_a"], r["score_b"]) if ca < cb else (cb, ca, r["score_b"], r["score_a"])
        h2h.append(dict(a=a, b=b, date=r["date"], tournament=r["tournament"], stage=r["stage"], sa=sa, sb=sb, bo=2 * max(sa, sb) - 1, patch=r.get("patch"), year=_year(r["date"])))
h2h.sort(key=lambda r: (r["date"][6:], r["date"][3:5], r["date"][:2]), reverse=True)
w("h2h", h2h)

# ---- matchups: every pairing at Bo1 / Bo3 / Bo5 ----
pg = J("data/matchups.json")["p_game"]
def _pser(p, bo):
    d = _math.log(p / (1 - p)); need = bo // 2 + 1
    return round(sum(v for k, v in _R.series_probs_mixed(d, bo, 1e-6).items() if int(k.split("-")[0]) == need), 4)
mu = []
for i, a in enumerate(sorted(WORLDS)):
    for b in sorted(WORLDS)[i + 1:]:
        p = pg[a][b]
        mu.append(dict(a=a, b=b, p_game=round(p, 4), p_bo3=_pser(p, 3), p_bo5=_pser(p, 5)))
w("matchups", mu)
print("h2h", len(h2h), "matchups", len(mu))
```

Note on `_pser`: `p_game` in `data/matchups.json` is already the marginal single-game probability (shock integrated out), so the series probability uses the sigma-free binomial (`1e-6`), the same quantity the dashboard's `p_series_a` was derived from for a fixed game probability.

/** Rows of data/dash/<name>.json, or [] when the file does not exist yet. */
export function load<T = Record<string, unknown>>(name: string): T[] {
  if (!cache.has(name)) {
    const p = resolve(DASH, `${name}.json`);
    cache.set(name, existsSync(p) ? (JSON.parse(readFileSync(p, 'utf8')) as T[]) : []);

```python
#!/usr/bin/env python3
"""Copy team logos and cached Riot art into site/public/art/ for the Astro build. Run after export_dashboard.py."""
import glob, json, os, re, shutil
ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
OUT = os.path.join(ROOT, "site", "public", "art")
nkey = lambda n: re.sub(r"[^a-z]", "", str(n).lower())

def copy_all(pattern, dest, rename=lambda f: f):
    os.makedirs(dest, exist_ok=True); n = 0
    for p in sorted(glob.glob(pattern)):
        shutil.copyfile(p, os.path.join(dest, rename(os.path.basename(p)))); n += 1
    return n

logos = copy_all(os.path.join(ROOT, "dashboard", "logos", "*.png"), os.path.join(OUT, "logos"))
extras = copy_all(os.path.join(ROOT, "dashboard", "extras", "*.png"), os.path.join(OUT, "extras"))
champs = copy_all(os.path.join(ROOT, "dashboard", "ddragon_cache", "cdn_*_img_champion_*.png"), os.path.join(OUT, "champ"), lambda f: f.split("_img_champion_")[1])
items = copy_all(os.path.join(ROOT, "dashboard", "ddragon_cache", "cdn_*_img_item_*.png"), os.path.join(OUT, "item"), lambda f: f.split("_img_item_")[1])
art = {nkey(f[:-4]): f for f in os.listdir(os.path.join(OUT, "champ"))}
art["ksante"] = art.get("ksante") or next((f for k, f in art.items() if k == "ksante"), None)
os.makedirs(os.path.join(ROOT, "site", "src", "lib"), exist_ok=True)
json.dump(art, open(os.path.join(ROOT, "site", "src", "lib", "art.json"), "w", encoding="utf-8"), indent=0, sort_keys=True)
print(f"logos {logos} extras {extras} champions {champs} items {items}")
```
